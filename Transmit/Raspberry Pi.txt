Raspberry Pi Steps for Apex Rover
Dual Camera Server + Sensor Bridge

الهدف:
تشغيل Raspberry Pi عشان يعمل شيئين:

1. بث كاميرتين للموبايل:

   * Front Camera للروبوت
   * Arm Camera للذراع

2. نقل بيانات السنسورات:

   * Arduino Mega يرسل بيانات السنسورات إلى Raspberry Pi
   * Raspberry Pi يرسلها إلى ESP32
   * ESP32 يبثها إلى تطبيق الموبايل عبر WebSocket

مهم:
Raspberry Pi لا يتحكم بحركة الروبوت.
كل التحكم Manual فقط من الموبايل:

Mobile App → ESP32 → Mega / UNO

أما Raspberry Pi وظيفته فقط:

Camera Stream + Sensor Bridge

```
def open(self):
    if self.cap is not None:
        self.cap.release()

    print(f"[INFO] Opening {self.name}: {self.device}")

    self.cap = cv2.VideoCapture(self.device, cv2.CAP_V4L2)
    self.cap.set(cv2.CAP_PROP_FRAME_WIDTH, FRAME_WIDTH)
    self.cap.set(cv2.CAP_PROP_FRAME_HEIGHT, FRAME_HEIGHT)
    self.cap.set(cv2.CAP_PROP_FPS, 20)

    self.opened = self.cap.isOpened()

    if self.opened:
        print(f"[OK] {self.name} opened")
    else:
        print(f"[ERROR] Could not open {self.name}")

    return self.opened

def read_frame(self):
    if self.cap is None or not self.cap.isOpened():
        self.open()

    if self.cap is None or not self.cap.isOpened():
        return None

    ret, frame = self.cap.read()

    if not ret or frame is None:
        print(f"[WARN] Failed to read from {self.name}, reopening...")
        self.open()
        return None

    return frame

def generate_mjpeg(self):
    while True:
        frame = self.read_frame()

        if frame is None:
            time.sleep(0.1)
            continue

        ok, buffer = cv2.imencode(
            ".jpg",
            frame,
            [int(cv2.IMWRITE_JPEG_QUALITY), JPEG_QUALITY],
        )

        if not ok:
            continue

        jpg = buffer.tobytes()

        yield (
            b"--frame\r\n"
            b"Content-Type: image/jpeg\r\n\r\n" + jpg + b"\r\n"
        )

        time.sleep(0.03)

def snapshot(self):
    frame = self.read_frame()

    if frame is None:
        return None

    ok, buffer = cv2.imencode(
        ".jpg",
        frame,
        [int(cv2.IMWRITE_JPEG_QUALITY), JPEG_QUALITY],
    )

    if not ok:
        return None

    return buffer.tobytes()
```

11) إضافة بث بيانات السنسورات للموبايل
======================================

بعد تشغيل بث الكاميرات، صار لازم نشغل كمان جسر السنسورات.

الفيديو يشتغل من:

Raspberry Pi -> Mobile App

أما السنسورات تشتغل بهذا المسار:

Arduino Mega -> Raspberry Pi -> ESP32 -> Mobile App

الموبايل ينتظر رسالة WebSocket من ESP32 بهذا الشكل:

SENSOR:PITCH=2.4;ROLL=-1.1;FRONT=35.6;REAR=18.2;BALANCE=STABLE

أو:

SENSOR:PITCH=16.2;ROLL=4.1;FRONT=12.0;REAR=8.5;BALANCE=WARNING

أو:

SENSOR:PITCH=31.5;ROLL=27.2;FRONT=4.0;REAR=5.5;BALANCE=DANGER

لذلك لازم Raspberry Pi يقرأ السنسورات من Mega ويرسلها للـ ESP32 عن طريق endpoint:

http://192.168.4.1/sensor_update

مهم جدًا:
Mega في الخطة النهائية يرسل السنسورات تلقائيًا بهذا الشكل:

SENSOR:PITCH=2.30;ROLL=-1.10;UF=8.50;UR=9.20;ALERT=NONE

لذلك sensor_bridge.py لازم يقرأ SENSOR مباشرة.
لا نستخدم GET:SENSORS.
لا ننتظر DATA:.

import time
import serial
import requests

# ==================================================

# Apex Rover Sensor Bridge

#

# Flow:

# Arduino Mega -> Raspberry Pi -> ESP32 -> Mobile App

#

# Mega sends automatically:

# SENSOR:PITCH=2.30;ROLL=-1.10;UF=8.50;UR=9.20;ALERT=NONE

#

# ESP32 receives:

# /sensor_update?pitch=...&roll=...&front=...&rear=...&balance=...

# ==================================================

ESP32_URL = "http://192.168.4.1"

# عدّل هذا البورت حسب نتيجة:

# ls /dev/ttyACM*

# أو:

# ls /dev/ttyUSB*

MEGA_PORT = "/dev/ttyACM0"

BAUD_RATE = 9600
FORWARD_TIMEOUT = 0.5

def parse_sensor_line(line):
line = line.strip()

```
if not line.startswith("SENSOR:"):
    return None

payload = line[len("SENSOR:"):]
parts = payload.split(";")

data = {}

for part in parts:
    if "=" not in part:
        continue

    key, value = part.split("=", 1)
    key = key.strip().upper()
    value = value.strip()
    data[key] = value

pitch = float(data.get("PITCH", 0.0))
roll = float(data.get("ROLL", 0.0))
front = float(data.get("UF", data.get("FRONT", -1.0)))
rear = float(data.get("UR", data.get("REAR", -1.0)))
alert = data.get("ALERT", "NONE").upper()

balance = compute_balance_status(pitch, roll, alert)

return {
    "pitch": pitch,
    "roll": roll,
    "front": front,
    "rear": rear,
    "balance": balance,
    "alert": alert,
}
```

def compute_balance_status(pitch, roll, alert):
abs_pitch = abs(pitch)
abs_roll = abs(roll)

```
if alert == "TILT_DANGER":
    return "DANGER"

if alert in ("OBSTACLE_FRONT", "OBSTACLE_REAR"):
    return "WARNING"

if abs_pitch >= 30 or abs_roll >= 25:
    return "DANGER"

if abs_pitch >= 15 or abs_roll >= 12:
    return "WARNING"

return "STABLE"
```

def send_to_esp32(sensor_data):
try:
response = requests.get(
f"{ESP32_URL}/sensor_update",
params={
"pitch": f"{sensor_data['pitch']:.2f}",
"roll": f"{sensor_data['roll']:.2f}",
"front": f"{sensor_data['front']:.2f}",
"rear": f"{sensor_data['rear']:.2f}",
"balance": sensor_data["balance"],
},
timeout=FORWARD_TIMEOUT,
)

```
    print("[ESP32]", response.text)

except requests.RequestException as e:
    print("[ERROR] Could not send to ESP32:", e)
```

def main():
print("====================================")
print("Apex Rover Sensor Bridge")
print(f"Mega Port : {MEGA_PORT}")
print(f"ESP32 URL : {ESP32_URL}")
print("====================================")

```
while True:
    try:
        with serial.Serial(MEGA_PORT, BAUD_RATE, timeout=0.5) as mega:
            time.sleep(2)
            mega.reset_input_buffer()

            print("[OK] Connected to Mega")
            print("[INFO] Waiting for SENSOR lines from Mega...")

            while True:
                line = mega.readline().decode(errors="ignore").strip()

                if line:
                    print("[MEGA]", line)

                sensor_data = parse_sensor_line(line)

                if sensor_data is not None:
                    print("[SENSOR]", sensor_data)
                    send_to_esp32(sensor_data)

    except serial.SerialException as e:
        print("[ERROR] Mega serial problem:", e)
        print("[INFO] Retrying in 2 seconds...")
        time.sleep(2)
```

19) ملاحظات مهمة
================

1. Raspberry Pi لا يتحكم بحركة الروبوت.
2. التحكم فقط من الموبايل.
3. Raspberry Pi يستخدم فقط للكاميرات والسنسورات.
4. Mega يرسل SENSOR lines إلى Raspberry Pi.
5. ESP32 يستقبل sensor_update ويبثها للموبايل.
6. إذا تغير IP الرازبيري، لازم نعدل IP داخل Flutter.
7. إذا لم تظهر بيانات السنسورات، افحص:

   * كود ESP32 مرفوع وفيه /sensor_update
   * كود Mega يرسل SENSOR:
   * البورت MEGA_PORT صحيح
   * Raspberry متصل على Apex_Rover_Net
   * الموبايل متصل WebSocket مع ESP32
